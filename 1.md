```
    return round($x / $y, 3);
};

{
    if ($n <= 1) return 1;
    return $n * factorial($n - 1);
}
```

# Задание 7.1
```
<?php
function logFile()
{
    $systemFile = "systemfile.log";
    $visitTime = date('Y-m-d H:i:s');
    $ip = $_SERVER['REMOTE_ADDR'];
    $text = $visitTime . " - IP: " . $ip . "\n";
    echo file_put_contents($systemFile, $text, FILE_APPEND);
}
function readlogFile()
{
    $systemFile = "systemfile.log";
    if (file_exists($systemFile)) {
        $content = file_get_contents($systemFile);
        echo "<pre>:Журнал посещений: \n $content </pre>";
    } else {
        echo "Файла не существует\n";
    }
}
logFile();
readlogFile();
?>
```
